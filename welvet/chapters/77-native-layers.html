Chapter 77

NativeCPU — TrainNativeMSE per layer

github.com/openfluke/welvet/layers✅ signed kinds

Why it exists

Each signed layer has a packed C ABI NormalBP path. One smoke walks Dense → CNN/ConvT → norms → RNN/LSTM → Sequential/Residual → Parallel → Embedding → Softmax → KMeans → MHA/SwiGLU → Mamba/GDN/Metacognition with Capture + Sync.

What it is

Set Exec.Backend = BackendNativeCPU, call package TrainNativeMSE, then SyncWeightsFromNative when Go Stores must match C++ (entity export).

Opt-in / Sync semantics: §76. Mode-aware stack/parallel/mesh/line: §78. Example tree: example/welvet/77-native-layers.

func main() {
	if !nativecpp.NativeCPUAvailable() {
		fmt.Println("SKIP: need linux/amd64 libwelvet")
		return
	}
	demos := []struct {
		name string
		fn   func() error
	}{
		{"dense", demoDense},
		{"rmsnorm", demoRMS},
		{"layernorm", demoLN},
		{"swiglu", demoSwiGLU},
		{"mha", demoMHA},
		{"softmax", demoSoftmax},
		{"embedding", demoEmbed},
		{"cnn1", demoCNN1},
		{"cnn2", demoCNN2},
		{"cnn3", demoCNN3},
		{"convt1", demoConvT1},
		{"convt2", demoConvT2},
		{"convt3", demoConvT3},
		{"rnn", demoRNN},
		{"lstm", demoLSTM},
		{"sequential", demoSeq},
		{"residual", demoRes},
		{"parallel", demoPara},
		{"kmeans", demoKMeans},
		{"mamba", demoMamba},
		{"gdn", demoGDN},
		{"metacognition", demoMeta},
	}
	ok := 0
	for _, d := range demos {
		if err := d.fn(); err != nil {
			fmt.Println("FAIL", d.name, err)
			continue
		}
		fmt.Println("OK  ", d.name)
		ok++
	}
	fmt.Printf("native-layers ok=%d/%d\n", ok, len(demos))
	if ok != len(demos) {
		panic("native-layers: some demos failed")
	}
}

func fill(shape ...int) *core.Tensor[float32] {
	t := core.NewTensor[float32](shape...)
	for i := range t.Data {
		t.Data[i] = 0.1 + float32(i%7)*0.03
	}
	return t
}

func tgtLike(x *core.Tensor[float32]) *core.Tensor[float32] {
	t := core.NewTensor[float32](x.Shape...)
	for i := range t.Data {
		t.Data[i] = 0.15
	}
	return t
}

func demoDense() error {
	l, err := dense.New(4, 4, core.ActivationLinear, core.DTypeFloat32)
	if err != nil {
		return err
	}
	l.Exec.Backend = core.BackendNativeCPU
	x := fill(2, 4)
	_, err = dense.TrainNativeMSE(l, x, tgtLike(x), 0.05)
	if err != nil {
		return err
	}
	return l.SyncWeightsFromNative()
}

func demoPara() error {
	init := make([]float32, 2*4*4)
	for i := range init {
		init[i] = 0.03 * float32(i+1)
	}
	l, err := parallel.NewConfigured(parallel.Config{Dim: 4, OutFeat: 4, Branches: 2, Combine: parallel.CombineAdd},
		core.DTypeFloat32, quant.FormatNone, init, nil)
	if err != nil {
		return err
	}
	l.Exec.Backend = core.BackendNativeCPU
	l.SyncBranchExec()
	x := fill(2, 4)
	_, err = parallel.TrainNativeMSE(l, x, tgtLike(x), 0.05)
	if err != nil {
		return err
	}
	return l.SyncWeightsFromNative()
}
